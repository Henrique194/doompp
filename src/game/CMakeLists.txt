project("game" VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)

set(PACKAGE_NAME ${PROJECT_NAME})
set(PACKAGE_TARNAME "game")
set(PACKAGE_VERSION ${PROJECT_VERSION})
set(PACKAGE_STRING "${PROJECT_NAME} ${PROJECT_VERSION}")

configure_file(${CONFIG_H_IN} "${CMAKE_CURRENT_BINARY_DIR}/config.h")

set(SRC_FILES main.c)
add_library(${PACKAGE_TARNAME} SHARED ${SRC_FILES})
target_include_directories(${PACKAGE_TARNAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
set_target_properties(${PACKAGE_TARNAME} PROPERTIES PREFIX "")
